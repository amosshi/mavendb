1   package org.mavendb.api;
2   
3   import org.apache.catalina.filters.RateLimitFilter;
4   import org.apache.catalina.util.ExactRateLimiter;
5   import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
6   import org.springframework.boot.web.servlet.FilterRegistrationBean;
7   import org.springframework.context.annotation.Bean;
8   import org.springframework.context.annotation.Configuration;
9   import org.springframework.core.Ordered;
10  
11  /**
12   * Limits requests per client IP with Tomcat's built-in {@link RateLimitFilter}, so one client cannot use up a
13   * small VPS. Over the limit the answer is <code>429 Too Many Requests</code>.
14   *
15   * <p>The client IP is the request's remote address. Behind a reverse proxy on the same host,
16   * <code>server.forward-headers-strategy=native</code> (in <code>application.properties</code>) makes Tomcat take it from
17   * <code>X-Forwarded-For</code>, before this filter runs; that header is only trusted from internal addresses.
18   *
19   * <p>Only the API paths are limited: <code>/actuator/health</code> stays reachable for monitoring.
20   * The number of requests served at the same time is capped by <code>server.tomcat.threads.max</code>.
21   */
22  @Configuration(proxyBeanMethods = false)
23  @ConditionalOnProperty(prefix = "mavendb.api.rate-limit", name = "enabled", havingValue = "true", matchIfMissing = true)
24  public class RateLimitConfig {
25  
26      /** Created by Spring. */
27      public RateLimitConfig() {
28          // Nothing to set up
29      }
30  
31      /** Length of a counting window, in seconds. */
32      static final int WINDOW_SECONDS = 60;
33  
34      /**
35       * @param props API settings
36       * @return Tomcat's rate limit filter on <code>/v1/*</code>, ahead of the other filters
37       */
38      @Bean
39      FilterRegistrationBean<RateLimitFilter> rateLimitFilter(ApiProperties props) {
40          FilterRegistrationBean<RateLimitFilter> registration = new FilterRegistrationBean<>(new RateLimitFilter());
41          registration.addUrlPatterns("/v1/*");
42          registration.setOrder(Ordered.HIGHEST_PRECEDENCE + 10);
43          registration.addInitParameter("bucketDuration", Integer.toString(WINDOW_SECONDS));
44          registration.addInitParameter("bucketRequests", Integer.toString(props.rateLimit().perClientPerMinute()));
45          // The default FastRateLimiter rounds the window up to a power of two (64 s); keep it at exactly one minute
46          registration.addInitParameter("rateLimitClassName", ExactRateLimiter.class.getName());
47          // RateLimit and RateLimit-Policy headers (IETF draft): the client sees its quota and the requests left
48          registration.addInitParameter("exposeHeaders", "true");
49          // Name of the policy in those headers: RateLimit-Policy: "per-ip";q=60;w=60
50          registration.addInitParameter("policyName", "per-ip");
51          return registration;
52      }
53  }
