1   package org.mavendb.api;
2   
3   import java.time.Duration;
4   import java.util.List;
5   import java.util.Locale;
6   import java.util.regex.Pattern;
7   import org.springframework.http.CacheControl;
8   import org.springframework.http.HttpStatus;
9   import org.springframework.http.ResponseEntity;
10  import org.springframework.web.bind.annotation.GetMapping;
11  import org.springframework.web.bind.annotation.PathVariable;
12  import org.springframework.web.bind.annotation.RequestMapping;
13  import org.springframework.web.bind.annotation.RequestParam;
14  import org.springframework.web.bind.annotation.RestController;
15  import org.springframework.web.server.ResponseStatusException;
16  
17  /**
18   * Metadata queries, version 1. Lookups by key only: no free text search, so every request is an index lookup.
19   */
20  @RestController
21  @RequestMapping("/v1")
22  public class GavController {
23  
24      /** Longest accepted group or artifact id; the schema allows 256 (longest in Maven Central: 129). */
25      static final int MAX_ID_LENGTH = 256;
26      /** Longest accepted file name. */
27      static final int MAX_FILE_NAME_LENGTH = 512;
28      /** Default and largest page of {@link #artifacts}. */
29      static final int DEFAULT_PAGE_SIZE = 100;
30      static final int MAX_PAGE_SIZE = 1000;
31  
32      private static final Pattern SHA1 = Pattern.compile("[0-9a-f]{40}");
33  
34      /** The data only changes when the SQLite file is replaced, so clients and proxies may cache answers. */
35      private static final CacheControl CACHE = CacheControl.maxAge(Duration.ofHours(1)).cachePublic();
36  
37      private final GavRepository repository;
38  
39      GavController(GavRepository repository) {
40          this.repository = repository;
41      }
42  
43      /** Which GAV has this file, like <code>commons-lang3-3.12.0.jar</code>. */
44      @GetMapping("/files/{fileName}")
45      ResponseEntity<List<Gav>> byFileName(@PathVariable String fileName) {
46          checkLength("fileName", fileName, MAX_FILE_NAME_LENGTH);
47          return found(this.repository.findByFileName(fileName));
48      }
49  
50      /** Which GAV has a file with this SHA-1. */
51      @GetMapping("/sha1/{sha1}")
52      ResponseEntity<List<Gav>> bySha1(@PathVariable String sha1) {
53          String normalized = sha1.toLowerCase(Locale.ROOT);
54          if (!SHA1.matcher(normalized).matches()) {
55              throw badRequest("sha1 must be 40 hex characters");
56          }
57          return found(this.repository.findBySha1(normalized));
58      }
59  
60      /** All artifacts of a group, sorted by artifact id, in pages. */
61      @GetMapping("/groups/{groupId}/artifacts")
62      ResponseEntity<List<Artifact>> artifacts(@PathVariable String groupId,
63                                               @RequestParam(defaultValue = "" + DEFAULT_PAGE_SIZE) int limit,
64                                               @RequestParam(defaultValue = "0") int offset) {
65          checkLength("groupId", groupId, MAX_ID_LENGTH);
66          if (limit < 1 || limit > MAX_PAGE_SIZE) {
67              throw badRequest("limit must be 1 to " + MAX_PAGE_SIZE);
68          }
69          if (offset < 0) {
70              throw badRequest("offset must not be negative");
71          }
72          List<Artifact> page = this.repository.findArtifacts(groupId, limit, offset);
73          // An empty first page means the group does not exist; an empty later page is just the end of the list
74          return offset == 0 ? found(page) : ResponseEntity.ok().cacheControl(CACHE).body(page);
75      }
76  
77      /** The latest version of an artifact. */
78      @GetMapping("/groups/{groupId}/artifacts/{artifactId}/latest")
79      ResponseEntity<LatestVersion> latest(@PathVariable String groupId, @PathVariable String artifactId) {
80          checkLength("groupId", groupId, MAX_ID_LENGTH);
81          checkLength("artifactId", artifactId, MAX_ID_LENGTH);
82          return this.repository.findLatest(groupId, artifactId)
83              .map(v -> ResponseEntity.ok().cacheControl(CACHE).body(v))
84              .orElseThrow(() -> new ResponseStatusException(HttpStatus.NOT_FOUND, "artifact not found"));
85      }
86  
87      private static <T> ResponseEntity<List<T>> found(List<T> rows) {
88          if (rows.isEmpty()) {
89              throw new ResponseStatusException(HttpStatus.NOT_FOUND, "not found");
90          }
91          return ResponseEntity.ok().cacheControl(CACHE).body(rows);
92      }
93  
94      private static void checkLength(String name, String value, int max) {
95          if (value.length() > max) {
96              throw badRequest(name + " is longer than " + max + " characters");
97          }
98      }
99  
100     private static ResponseStatusException badRequest(String reason) {
101         return new ResponseStatusException(HttpStatus.BAD_REQUEST, reason);
102     }
103 }
