20  /**
21   * Minimal OpenSearch REST client for the bulk load, on the JDK {@link HttpClient}: no client library needed.
22   *
23   * <p>Load cycle of one index:
24   * <ol>
25   *   <li>{@link #recreateIndex(String, String)}: drop the index and create it from <code>db/opensearch/create.json</code>,
26   *       tuned for loading: no replicas, refresh disabled, asynchronous translog</li>
27   *   <li>{@link #bulk(String, byte[], int, long)}: one {@code _bulk} request per batch, from several writers</li>
28   *   <li>{@link #finishLoad(String)}: restore search settings and refresh, so all documents become searchable</li>
29   * </ol>
30   *
31   * <p>Thread safe: {@link HttpClient} is shared by all writers.
32   */
33  final class OpenSearchClient implements AutoCloseable {
34  
35      private static final Logger LOG = Logger.getLogger(OpenSearchClient.class.getName());
36  
37      /** Generous: a bulk request of a large batch, or a refresh of the whole index, takes minutes on a busy node. */
38      private static final Duration REQUEST_TIMEOUT = Duration.ofMinutes(30);
39  
40      /** Compact JSON with plain numbers: {@code 42}, not {@code {"$numberLong": "42"}}. */
41      private static final JsonWriterSettings JSON_SETTINGS = JsonWriterSettings.builder().outputMode(JsonMode.RELAXED).build();
42  
43      /** Bulk responses list every item; keep only the error flag and failed items, so a good response is tiny. */
44      private static final String BULK_FILTER = "?filter_path=errors,items.*.error";
45  
46      /**
47       * Largest {@code _bulk} request, well below {@code http.max_content_length} (200mb in compose-opensearch.yml,
48       * 100mb by default): a batch of 10,000 large documents reached over 200 MB. Smaller requests also spare the node heap.
49       */
50      static final int MAX_BULK_REQUEST_BYTES = 50 * 1024 * 1024;
51  
52      private static final int ERROR_EXCERPT_LENGTH = 1_000;
53  
54      private final String baseUrl;
55      private final String authorization;
56      private final HttpClient http;
57  
58      /**
59       * Create a client.
60       *
61       * @param baseUrl Server URL, like {@code http://localhost:9200}
62       * @param user User for basic authentication; blank for a server without the security plugin
63       * @param password Password for basic authentication
64       */
65      OpenSearchClient(String baseUrl, String user, String password) {
66          this.baseUrl = baseUrl.endsWith("/") ? baseUrl.substring(0, baseUrl.length() - 1) : baseUrl;
67          this.authorization = (user == null || user.isBlank()) ? null
68              : "Basic " + Base64.getEncoder().encodeToString((user + ":" + password).getBytes(StandardCharsets.UTF_8));
69          this.http = HttpClient.newBuilder()
70              .version(HttpClient.Version.HTTP_1_1)
71              .connectTimeout(Duration.ofSeconds(30))
72              .build();
73      }
74  
75      /**
76       * Encode documents as a {@code _bulk} request body (NDJSON): an index action with the document id, then the source.
77       *
78       * <p>The {@code _id} field becomes the OpenSearch document id: it is metadata and must not be in the source.
79       * Thread safe: called on the conversion threads, so the writer only sends bytes.
80       *
81       * @param documents Documents with an {@code _id} field; the field is removed from each document
82       * @return Request body, UTF-8
83       */
84      static byte[] toBulkData(List<Document> documents) {
85          StringBuilder sb = new StringBuilder(documents.size() * 1_024);
86          for (Document document : documents) {
87              Object id = document.remove(DatabaseRepository.JSON_FIELD_ID);
88              sb.append("{\"index\":{\"_id\":\"").append(id).append("\"}}\n");
89              sb.append(document.toJson(JSON_SETTINGS)).append('\n');
90          }
91          return sb.toString().getBytes(StandardCharsets.UTF_8);
92      }
93  
94      /**
95       * Drop the index if it exists, then create it with the given settings and mappings.
96       *
97       * @param index Index name
98       * @param definition Index settings and mappings, JSON
99       */
100     void recreateIndex(String index, String definition) throws IOException {
101         HttpResponse<String> deleted = send(request("/" + index).DELETE(), "DELETE index " + index);
102         if (deleted.statusCode() != 200 && deleted.statusCode() != 404) {
103             throw failure("DELETE index " + index, deleted);
104         }
105         expectOk(send(json("/" + index, definition, "PUT"), "PUT index " + index), "PUT index " + index);
106         LOG.log(Level.INFO, "OpenSearch index {0} created", index);
107     }

109     /**
110      * Index one batch with a {@code _bulk} request.
111      *
112      * @param index Index name
113      * @param bulkData Request body, see {@link #toBulkData(List)}
114      * @param size Number of documents, for logging
115      * @param counter Record counter, for logging
116      * @throws IOException if the request fails, or any document is rejected
117      */
118     void bulk(String index, byte[] bulkData, int size, long counter) throws IOException {
119         for (int[] range : splitBulk(bulkData, MAX_BULK_REQUEST_BYTES)) {
120             HttpRequest.Builder request = request("/" + index + "/_bulk" + BULK_FILTER)
121                 .header("Content-Type", "application/x-ndjson")
122                 .POST(HttpRequest.BodyPublishers.ofByteArray(bulkData, range[0], range[1] - range[0]));
123             HttpResponse<String> response = send(request, "bulk");
124             expectOk(response, "bulk");
125 
126             // OpenSearch answers 200 even when documents are rejected; the errors flag tells
127             if (Boolean.TRUE.equals(Document.parse(response.body()).getBoolean("errors"))) {
128                 throw new IOException("OpenSearch rejected documents of records counter=" + counter
129                     + ", batchSize=" + size + ": " + excerpt(response.body()));
130             }
131         }
132     }
133 
134     /**
135      * Split a bulk body into requests of at most {@code maxBytes}, at document boundaries (every second line).
136      *
137      * <p>A batch has a fixed number of documents, but documents vary from 1 KB to over 1 MB
138      * (a long {@code Export-Package}), so a batch of large documents can pass {@code http.max_content_length}
139      * and fail with HTTP 413. A single document larger than {@code maxBytes} is sent alone.
140      * Splitting on the raw {@code '\n'} byte is safe: JSON escapes newlines in strings,
141      * and no UTF-8 multi-byte sequence contains that byte.
142      *
143      * @param data Bulk body, see {@link #toBulkData(List)}
144      * @param maxBytes Request size limit
145      * @return Ranges {@code [start, end)} covering all of {@code data}, in order
146      */
147     static List<int[]> splitBulk(byte[] data, int maxBytes) {
148         List<int[]> ranges = new ArrayList<>();
149         int start = 0;
150         int lastBoundary = 0;
151         boolean actionLine = true;
152         for (int i = 0; i < data.length; i++) {
153             if (data[i] != '\n') {
154                 continue;
155             }
156             actionLine = !actionLine;
157             if (!actionLine) {
158                 continue;   // end of the action line; the document ends after its source line
159             }
160             int boundary = i + 1;
161             if (boundary - start > maxBytes && lastBoundary > start) {
162                 ranges.add(new int[] {start, lastBoundary});
163                 start = lastBoundary;
164             }
165             lastBoundary = boundary;
166         }
167         if (start < data.length) {
168             ranges.add(new int[] {start, data.length});
169         }
170         return ranges;
171     }

173     /**
174      * Make the loaded index ready for searching: enable refresh and durable translog writes, then refresh.
175      *
176      * @param index Index name
177      */
178     void finishLoad(String index) throws IOException {
179         String settings = """
180             { "index": { "refresh_interval": "1s", "translog.durability": "request" } }""";
181         expectOk(send(json("/" + index + "/_settings", settings, "PUT"), "PUT settings"), "PUT settings");
182         expectOk(send(request("/" + index + "/_refresh").POST(HttpRequest.BodyPublishers.noBody()), "refresh"), "refresh");
183         LOG.log(Level.INFO, "OpenSearch index {0} refreshed", index);
184     }

186     /**
187      * Count the documents of an index.
188      *
189      * @param index Index name
190      * @return Number of documents
191      */
192     long count(String index) throws IOException {
193         HttpResponse<String> response = send(request("/" + index + "/_count").GET(), "count");
194         expectOk(response, "count");
195         return Document.parse(response.body()).get("count", Number.class).longValue();
196     }
197 
198     /** Bytes the index takes on disk, primaries and replicas: {@code _all.total.store.size_in_bytes}. */
199     long storeSizeBytes(String index) throws IOException {
200         HttpResponse<String> response = send(request("/" + index + "/_stats/store").GET(), "stats");
201         expectOk(response, "stats");
202         Document total = Document.parse(response.body()).get("_all", Document.class).get("total", Document.class);
203         return total.get("store", Document.class).get("size_in_bytes", Number.class).longValue();
204     }

206     private HttpRequest.Builder request(String path) {
207         HttpRequest.Builder builder = HttpRequest.newBuilder(URI.create(this.baseUrl + path)).timeout(REQUEST_TIMEOUT);
208         if (this.authorization != null) {
209             builder.header("Authorization", this.authorization);
210         }
211         return builder;
212     }
213 
214     private HttpRequest.Builder json(String path, String body, String method) {
215         return request(path)
216             .header("Content-Type", "application/json")
217             .method(method, HttpRequest.BodyPublishers.ofString(body, StandardCharsets.UTF_8));
218     }
219 
220     private HttpResponse<String> send(HttpRequest.Builder request, String what) throws IOException {
221         try {
222             return this.http.send(request.build(), HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
223         } catch (InterruptedException e) {
224             Thread.currentThread().interrupt();
225             throw new IOException("Interrupted: OpenSearch " + what, e);
226         }
227     }
228 
229     private static void expectOk(HttpResponse<String> response, String what) throws IOException {
230         if (response.statusCode() != 200) {
231             throw failure(what, response);
232         }
233     }
234 
235     private static IOException failure(String what, HttpResponse<String> response) {
236         return new IOException("OpenSearch " + what + " failed with HTTP " + response.statusCode() + ": " + excerpt(response.body()));
237     }
238 
239     private static String excerpt(String body) {
240         return body.length() <= ERROR_EXCERPT_LENGTH ? body : body.substring(0, ERROR_EXCERPT_LENGTH) + "...";
241     }
242 
243     @Override
244     public void close() {
245         this.http.close();
246     }
