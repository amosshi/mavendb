3   import java.io.File;

5   /**
6    * Locations of the application folders, resolved next to the jar file.
7    *
8    * <p>Depends only on {@link DatabaseType}.
9    */
10  final class AppPaths {
11  
12      /** Directory for configuration files. */
13      static final String DIR_ETC = "etc";
14  
15      /** Directory for log files. */
16      static final String DIR_LOG = "log";
17  
18      /** Directory for database scripts; each database type has a sub folder, like <code>db/mysql</code>. */
19      static final String DIR_DB = "db";
20  
21      private AppPaths() {
22      }
23  
24      /**
25       * Get the full path of a file in an application folder.
26       *
27       * @param dir Directory name, like {@link #DIR_ETC}, {@link #DIR_LOG}, <code>db/mysql</code>
28       * @param file File name, like <code>config.properties</code>, <code>create.sql</code>
29       * @return The full path of the file
30       */
31      static String getFileName(String dir, String file) {
32          File baseDir = new File(AppPaths.class.getProtectionDomain().getCodeSource().getLocation().getPath());
33          return baseDir.getParent() + File.separator + dir + File.separator + file;
34      }
35  
36      /**
37       * Get the full path of a database script.
38       *
39       * @param dbType Database type, selects the <code>db/&lt;dbtype&gt;</code> folder
40       * @param script Script file name, like <code>create.sql</code>
41       * @return The full path of the script
42       */
43      static String getDbScriptFileName(DatabaseType dbType, String script) {
44          return getFileName(DIR_DB + File.separator + dbType.name().toLowerCase(), script);
45      }
46  }
