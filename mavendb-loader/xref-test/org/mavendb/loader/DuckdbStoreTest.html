25  /**
26   * Tests {@link DatabaseRepository#storeDuckDB(List, long)} against a DuckDB file created by db/duckdb/create.sql.
27   */
28  @DisplayName("DuckDB Appender Tests")
29  class DuckdbStoreTest {
30  
31      private static final String[] TRICKY = {
32          "tab\there", "nl\nhere", "back\\slash", "\\N", "NULL", "emoji 😀 例子", "", "ctrl\u0001\u0000z"
33      };
34  
35      @TempDir
36      Path tempDir;
37  
38      private String createDatabase() throws IOException, SQLException {
39          String url = "jdbc:duckdb:" + this.tempDir.resolve("test.duckdb");
40          String script = Files.readString(Path.of("src/main/resources/db/duckdb/create.sql"), StandardCharsets.UTF_8);
41          try (Connection conn = DriverManager.getConnection(url); Statement stmt = conn.createStatement()) {
42              for (String sql : script.replaceAll("(?m)^--.*$", "").split(";")) {
43                  if (!sql.isBlank()) {
44                      stmt.execute(sql);
45                  }
46              }
47          }
48          return url;
49      }
50  
51      private static DatabaseRepository.SqlRow row(int i) {
52          DatabaseRepository.SqlRow row = new DatabaseRepository.SqlRow();
53          row.setLong(1, i + 1L);
54          row.setInt(2, 7);
55          row.setLong(3, 123_456_789_012L);
56          row.setObject(4, (long) i);
57          row.setObject(5, null);
58          row.setBoolean(7, true);
59          row.setBoolean(8, false);
60          row.setString(10, null);
61          row.setString(11, "g" + i);
62          row.setString(12, "a");
63          row.setString(13, "v");
64          for (int c = 14; c <= 29; c++) {
65              row.setString(c, TRICKY[i]);
66          }
67          row.setString(30, "{\"k\":\"say \\\"hi\\\"\",\"i\":" + i + "}");
68          return row;
69      }
70  
71      @Test
72      @DisplayName("Rows round trip through the Appender: numbers, booleans, nulls, special characters, JSON")
73      void testRoundTrip() throws Exception {
74          String url = createDatabase();
75          List<DatabaseRepository.SqlRow> rows = new ArrayList<>();
76          for (int i = 0; i < TRICKY.length; i++) {
77              rows.add(row(i));
78          }
79  
80          try (DatabaseRepository repo = new DatabaseRepository(DatabaseType.DUCKDB, url, new Properties())) {
81              repo.storeDuckDB(rows, rows.size());
82              // At least the header and one block of the file
83              assertTrue(repo.databaseSizeBytes("central") > 0, "database size");
84          }
85  
86          try (Connection conn = DriverManager.getConnection(url);
87               ResultSet rs = conn.createStatement().executeQuery("SELECT *, json->>'k' AS k FROM gav ORDER BY seqid")) {
88              int i = 0;
89              while (rs.next()) {
90                  assertEquals(i + 1L, rs.getLong("seqid"));
91                  assertEquals(7, rs.getInt("major_version"));
92                  assertEquals(123_456_789_012L, rs.getLong("version_seq"));
93                  assertEquals(i, rs.getLong("record_modified"));
94                  assertNull(rs.getObject("file_modified"));
95                  assertNull(rs.getObject("file_size"), "a parameter that is never set is null");
96                  assertTrue(rs.getBoolean("has_signature"));
97                  assertFalse(rs.getBoolean("has_sources"));
98                  assertNull(rs.getObject("sha1"));
99                  assertEquals("g" + i, rs.getString("group_id"));
100                 for (String col : List.of("classifier", "name", "description", "bundle_docurl", "export_package", "export_service")) {
101                     assertEquals(TRICKY[i], rs.getString(col), col + " of row " + i);
102                 }
103                 assertEquals("say \"hi\"", rs.getString("k"));
104                 i++;
105             }
106             assertEquals(TRICKY.length, i);
107         }
108     }
