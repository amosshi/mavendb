29      @Test
30      @DisplayName("Credentials are masked in connection URLs")
31      void testMaskUrl() {
32          assertEquals("mongodb://***@localhost:27017/admin", RunSummary.maskUrl("mongodb://root:secret@localhost:27017/admin"));
33          assertEquals("jdbc:mysql://h:3306/db?user=a&password=***&x=1", RunSummary.maskUrl("jdbc:mysql://h:3306/db?user=a&password=secret&x=1"));
34          assertEquals("jdbc:duckdb:mavendb.duckdb", RunSummary.maskUrl("jdbc:duckdb:mavendb.duckdb"));
35          assertNull(RunSummary.maskUrl(null));
36      }

38      @Test
39      @DisplayName("A successful run has durations, record counts and throughput")
40      void testSuccess() throws Exception {
41          RunSummary summary = new RunSummary(DatabaseType.MYSQL, "file:///index/", "20261114.120000-mysql.log");
42          summary.setDatabaseUrl("jdbc:mysql://127.0.0.1:3306/mavendb");
43          summary.setDatabaseSizeBytes(3L * 1024 * 1024 * 1024 / 2);
44          summary.putDuration("createSchema", 65);
45          summary.putDuration("import", 2_000);
46          summary.putDuration("refresh", 1_000);
47          summary.setRecords(1_003, 1_000);
48          summary.putTableRows("g", 10);
49          summary.putSetting("batchSize", 50_000);
50          summary.setMaxFieldLengths(Map.of("version", 145));
51          Instant published = Instant.now().minus(36, ChronoUnit.HOURS);
52          summary.setIndexTimestamp(published);
53          summary.addIndexChunk("nexus-maven-repository-index.gz", 1, published);
54          summary.succeeded();
55  
56          Path file = this.tempDir.resolve("run.json");
57          summary.write(file);
58          Document doc = Document.parse(Files.readString(file));
59  
60          assertEquals("SUCCESS", doc.getString("status"));
61          assertFalse(doc.containsKey("error"));
62          assertEquals("mysql", doc.getString("dbType"));
63          Document durations = doc.get("durationsMs", Document.class);
64          assertTrue(durations.containsKey("total"));
65          assertEquals(2_000, durations.get("import", Number.class).longValue());
66          Document records = doc.get("records", Document.class);
67          assertEquals(1_000, records.get("stored", Number.class).longValue());
68          assertEquals(3, records.get("skipped", Number.class).longValue());
69          assertEquals(500, records.get("importPerSecond", Number.class).longValue());
70          assertEquals(10, doc.get("tableRows", Document.class).get("g", Number.class).longValue());
71          assertEquals(145, doc.get("maxFieldLengths", Document.class).get("version", Number.class).intValue());
72          assertEquals("20261114.120000-mysql.log", doc.getString("logFile"));
73          Document database = doc.get("database", Document.class);
74          assertEquals(1_610_612_736L, database.get("sizeBytes", Number.class).longValue());
75          assertEquals(1.5, database.get("sizeGiB", Number.class).doubleValue());
76  
77          assertEquals(published.toString(), doc.getString("indexTimestamp"));
78          assertEquals(1.5, doc.get("indexAgeDays", Number.class).doubleValue());
79          List<Document> chunks = doc.get("index", Document.class).getList("chunks", Document.class);
80          assertEquals(1, chunks.size());
81          assertEquals("nexus-maven-repository-index.gz", chunks.get(0).getString("name"));
82          assertEquals(published.toString(), chunks.get(0).getString("timestamp"));
83      }

85      @Test
86      @DisplayName("A failed run records the error")
87      void testFailure() {
88          RunSummary summary = new RunSummary(DatabaseType.PSQL, "file:///index/", "x.log");
89          summary.failed(new IllegalStateException("boom"));
90          Document doc = summary.toDocument();
91          assertEquals("FAILED", doc.getString("status"));
92          assertEquals("java.lang.IllegalStateException: boom", doc.getString("error"));
93          assertFalse(doc.get("records", Document.class).containsKey("importPerSecond"));
94          assertTrue(doc.containsKey("indexTimestamp"), "always present, null when the index was not read");
95          assertNull(doc.get("indexTimestamp"));
96          assertFalse(doc.containsKey("indexAgeDays"));
97      }
