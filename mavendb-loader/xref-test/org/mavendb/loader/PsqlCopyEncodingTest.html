5   import java.nio.charset.StandardCharsets;
6   import java.util.List;

8   import org.junit.jupiter.api.DisplayName;

11  /**
12   * Unit tests for the COPY text encoding (PostgreSQL COPY, MySQL LOAD DATA) in {@link DatabaseRepository}.
13   */
14  @DisplayName("COPY / LOAD DATA encoding Tests")
15  class PsqlCopyEncodingTest {
16  
17      private static String escape(String value) {
18          StringBuilder sb = new StringBuilder();
19          DatabaseRepository.SqlRow.appendCopyEscaped(sb, value);
20          return sb.toString();
21      }
22  
23      @Test
24      @DisplayName("Plain text is copied unchanged")
25      void testPlainText() {
26          assertEquals("org.apache.maven", escape("org.apache.maven"));
27          assertEquals("", escape(""));
28      }
29  
30      @Test
31      @DisplayName("Backslash, newline, carriage return and tab are escaped")
32      void testSpecialCharacters() {
33          assertEquals("a\\\\b", escape("a\\b"));
34          assertEquals("line1\\nline2", escape("line1\nline2"));
35          assertEquals("a\\rb", escape("a\rb"));
36          assertEquals("col1\\tcol2", escape("col1\tcol2"));
37          assertEquals("\\\\\\n\\t", escape("\\\n\t"));
38      }
39  
40      @Test
41      @DisplayName("JSON escapes keep their backslash after COPY decoding")
42      void testJsonBackslash() {
43          assertEquals("{\"d\": \"say \\\\\"hi\\\\\"\"}", escape("{\"d\": \"say \\\"hi\\\"\"}"));
44      }
45  
46      @Test
47      @DisplayName("Non-ASCII text is encoded as UTF-8")
48      void testUnicode() {
49          DatabaseRepository.SqlRow row = new DatabaseRepository.SqlRow();
50          row.setString(11, "com.例子");
51          String line = new String(DatabaseRepository.toCopyData(List.of(row), DatabaseType.PSQL), StandardCharsets.UTF_8);
52          assertEquals("com.例子", line.split("\t", -1)[10]);
53      }
54  
55      @Test
56      @DisplayName("A row has 30 tab separated columns: null, numbers, booleans, strings")
57      void testRow() {
58          DatabaseRepository.SqlRow row = new DatabaseRepository.SqlRow();
59          row.setLong(1, 42L);
60          row.setInt(2, 3);
61          row.setLong(3, 300_000L);
62          row.setObject(4, null);
63          row.setBoolean(7, true);
64          row.setBoolean(8, false);
65          row.setString(11, "g\tx");
66          row.setString(30, "{\"k\":\"v\"}");
67  
68          String data = new String(DatabaseRepository.toCopyData(List.of(row, row), DatabaseType.PSQL), StandardCharsets.UTF_8);
69          String[] lines = data.split("\n", -1);
70          assertEquals(3, lines.length, "two rows, each ending with a newline");
71          assertEquals("", lines[2]);
72  
73          String[] cols = lines[0].split("\t", -1);
74          assertEquals(30, cols.length);
75          assertEquals("42", cols[0]);
76          assertEquals("3", cols[1]);
77          assertEquals("300000", cols[2]);
78          assertEquals("\\N", cols[3]);
79          assertEquals("\\N", cols[4]);
80          assertEquals("t", cols[6]);
81          assertEquals("f", cols[7]);
82          assertEquals("g\\tx", cols[10]);
83          assertEquals("{\"k\":\"v\"}", cols[29]);
84      }
85  
86      @Test
87      @DisplayName("MySQL LOAD DATA gets booleans as 1/0, everything else as for PostgreSQL")
88      void testMysqlRow() {
89          DatabaseRepository.SqlRow row = new DatabaseRepository.SqlRow();
90          row.setLong(1, 42L);
91          row.setBoolean(7, true);
92          row.setBoolean(8, false);
93          row.setString(11, "g\\x");
94  
95          String[] cols = new String(DatabaseRepository.toCopyData(List.of(row), DatabaseType.MYSQL), StandardCharsets.UTF_8)
96              .split("\n", -1)[0].split("\t", -1);
97          assertEquals(30, cols.length);
98          assertEquals("42", cols[0]);
99          assertEquals("1", cols[6]);
100         assertEquals("0", cols[7]);
101         assertEquals("\\N", cols[8]);
102         assertEquals("g\\\\x", cols[10]);
103     }
