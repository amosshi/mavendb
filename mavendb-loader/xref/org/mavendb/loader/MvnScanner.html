135     /**
136      * Factory method to safely create a MvnScanner instance.
137      * Validates the index folder path before object construction to prevent
138      * partially initialized objects vulnerable to finalizer attacks.
139      *
140      * @param folderPath The folder path to scan
141      * @param dbType The database type to use
142      * @return A validated MvnScanner instance
143      * @throws IllegalArgumentException if the path is invalid or contains suspicious patterns
144      */
145     public static MvnScanner create(String folderPath, DatabaseType dbType) throws IllegalArgumentException {
146         URI validatedUri = validateAndCreateURI(folderPath);
147         return new MvnScanner(validatedUri, dbType);
148     }

150     /**
151      * Validates the index folder path and converts it to a safe URI.
152      * Ensures the path is not null, empty, and doesn't contain path traversal attempts.
153      *
154      * @param folderPath The folder path to validate
155      * @return A validated URI
156      * @throws IllegalArgumentException if the path is invalid or contains suspicious patterns
157      */
158     private static URI validateAndCreateURI(String folderPath) throws IllegalArgumentException {
159         // Check for common path traversal patterns
160         if (folderPath.contains("..") || folderPath.contains("~")) {
161             throw new IllegalArgumentException("Index folder path contains suspicious patterns: " + folderPath);
162         }
163 
164         try {
165             URI uri = URI.create(folderPath);
166 
167             // Validate URI scheme for file:// URIs
168             if (uri.getScheme() != null && uri.getScheme().equals("file")) {
169                 // Normalize and validate file path
170                 Path path = Path.of(uri);
171                 // Ensure the path exists and is accessible
172                 if (!Files.exists(path)) {
173                     throw new IllegalArgumentException("Index folder path does not exist: " + folderPath);
174                 }
175                 if (!Files.isDirectory(path)) {
176                     throw new IllegalArgumentException("Index folder path is not a directory: " + folderPath);
177                 }
178                 if (!Files.isReadable(path)) {
179                     throw new IllegalArgumentException("Index folder path is not readable: " + folderPath);
180                 }
181             } else if (uri.getScheme() != null) {
182                 // For remote URIs (http, https, etc.), basic validation
183                 if (!uri.getScheme().matches("^[a-zA-Z][a-zA-Z0-9+.-]*$")) {
184                     throw new IllegalArgumentException("Invalid URI scheme: " + uri.getScheme());
185                 }
186             }
187 
188             return uri;
189         } catch (IllegalArgumentException e) {
190             throw e;
191         } catch (Exception e) {
192             throw new IllegalArgumentException("Invalid index folder path: " + folderPath, e);
193         }
194     }

197     /**
198      * Run the import: create the schema, scan the index into the database, then refresh the summary tables.
199      *
200      * @param config Configuration properties
201      * @param summary Run summary, filled in with durations, record counts and settings
202      */
203     public void perform(Properties config, RunSummary summary) throws IOException, SQLException {
204         this.configMgr = new ConfigurationManager(config);
205         this.summary = summary;
206 
207         int writers;
208         if (this.dbType == DatabaseType.SQLITE) {
209             LOG.log(Level.INFO, "SQLite only allows one writer, cannot write in multiple threads");
210             writers = 1;
211         } else {
212             writers = configMgr.getThreadPoolSize();
213         }
214         LOG.log(Level.INFO, "Concurrent database writers: {0}", writers);
215         summary.putSetting("writers", writers);
216         this.writerPermits = new Semaphore(writers);
217 
218         this.databaseRepository = switch (this.dbType) {
219             case MYSQL -> new DatabaseRepository(DatabaseType.MYSQL, configMgr.getMysqlUrl(), configMgr.mysqlConnectionProps);
220             case MARIADB -> new DatabaseRepository(DatabaseType.MARIADB, configMgr.getMariadbUrl(), configMgr.mariadbConnectionProps);
221             case PSQL -> new DatabaseRepository(DatabaseType.PSQL, configMgr.getPsqlUrl(), configMgr.psqlConnectionProps);
222             case SQLITE -> new DatabaseRepository(DatabaseType.SQLITE, configMgr.getSqliteUrl(), configMgr.sqliteConnectionProps);
223             case DUCKDB -> new DatabaseRepository(DatabaseType.DUCKDB, configMgr.getDuckdbUrl(), configMgr.duckdbConnectionProps);
224             case MONGODB -> new DatabaseRepository(configMgr.getMongodbUrl());
225             case OPENSEARCH -> new DatabaseRepository(configMgr.getOpensearchUrl(), configMgr.getOpensearchUser(), configMgr.getOpensearchPassword());
226         };
227         summary.setDatabaseUrl(switch (this.dbType) {
228             case MYSQL -> configMgr.getMysqlUrl();
229             case MARIADB -> configMgr.getMariadbUrl();
230             case PSQL -> configMgr.getPsqlUrl();
231             case SQLITE -> configMgr.getSqliteUrl();
232             case DUCKDB -> configMgr.getDuckdbUrl();
233             case MONGODB -> configMgr.getMongodbUrl();
234             case OPENSEARCH -> configMgr.getOpensearchUrl();
235         });
236         if (DatabaseRepository.SUPPORTED_SQL_DB_TYPES.contains(this.dbType)) {
237             long createStart = System.currentTimeMillis();
238             this.databaseRepository.executeSQLScript(DB_CREATE_SQL);
239             summary.putDuration("createSchema", System.currentTimeMillis() - createStart);
240         }
241 
242         try {
243             long start = System.currentTimeMillis();
244             try {
245                 // Returns only after every batch is written; a failed batch throws and stops the scan
246                 this.stepScan(writers);
247             } catch (RuntimeException e) {
248                 LOG.log(Level.SEVERE, "Scan failed, the database is incomplete, rerun the import", e);
249                 throw e;
250             }
251             LOG.log(Level.INFO, "Scan execution time={0}", System.currentTimeMillis() - start);
252             summary.putDuration("import", System.currentTimeMillis() - start);
253 
254             // All writers are done, release their pooled connections before the post-load scripts
255             this.databaseRepository.closeSQLConnections();
256 
257             // Refresh Data
258             long refreshStart = System.currentTimeMillis();
259             if (DatabaseRepository.SUPPORTED_SQL_DB_TYPES.contains(this.dbType)) {
260                 this.databaseRepository.executeSQLScript(DB_DATA_REFRESH_SQL);
261             } else if (this.dbType == DatabaseType.MONGODB) {
262                 this.databaseRepository.createIndexesMongoDB(this.indexId);
263             } else if (this.dbType == DatabaseType.OPENSEARCH) {
264                 this.databaseRepository.finishOpenSearch(this.indexId);
265             }
266             summary.putDuration("refresh", System.currentTimeMillis() - refreshStart);
267 
268             // Rows of the summary tables; gav has the stored record count.
269             // Informational only: a failed count must not fail a completed import.
270             if (DatabaseRepository.SUPPORTED_SQL_DB_TYPES.contains(this.dbType)) {
271                 for (String table : SUMMARY_TABLES) {
272                     try {
273                         summary.putTableRows(table, this.databaseRepository.countRows(table));
274                     } catch (SQLException e) {
275                         LOG.log(Level.WARNING, "Cannot count the rows of table " + table + " for the run summary", e);
276                     }
277                 }
278             } else if (this.dbType == DatabaseType.OPENSEARCH) {
279                 try {
280                     summary.putTableRows(DatabaseRepository.openSearchIndexName(this.indexId), this.databaseRepository.countOpenSearch(this.indexId));
281                 } catch (IOException e) {
282                     LOG.log(Level.WARNING, "Cannot count the documents of the OpenSearch index for the run summary", e);
283                 }
284             }
285 
286             // Informational only, like the row counts
287             try {
288                 summary.setDatabaseSizeBytes(this.databaseRepository.databaseSizeBytes(this.indexId));
289             } catch (SQLException | IOException | RuntimeException e) {
290                 LOG.log(Level.WARNING, "Cannot get the database size for the run summary", e);
291             }
292         } finally {
293             summary.setMaxFieldLengths(this.maxLengthTracker);
294             this.databaseRepository.close();
295         }
296 
297         LOG.log(Level.INFO, "Max length of the String fields: {0}{1}", new Object[] {System.lineSeparator(),
298                 this.maxLengthTracker.entrySet().stream()
299                 .map(entry -> entry.getKey() + "=" + entry.getValue())
300                 .collect(Collectors.joining(System.lineSeparator()))});
301     }

303     /**
304      * Scan maven index files and write them to the database in batches.
305      *
306      * <p>Pipeline (Java 24+ stream gatherers, JEP 485):
307      * <pre>
308      *   chunks → raw records (seq assigned) → windowFixed(batchSize)
309      *          → mapConcurrent(converters, convertBatch) → mapConcurrent(writers + 1, storeBatch)
310      * </pre>
311      * Reading and decoding the index is sequential and runs on this thread, which also assigns the
312      * record sequence in index order. Expanding records, building documents and preparing SQL rows is
313      * about two thirds of the CPU work, so it runs on {@code converters} virtual threads.
314      * {@code mapConcurrent} keeps the input order, so batches still reach the writers in sequence order.
315      *
316      * <p>Why {@code writers + 1} tasks plus {@link #writerPermits}: {@code mapConcurrent} blocks the scanning
317      * thread whenever all of its task slots are busy. With exactly {@code writers} slots, the next batch
318      * could only be handed over after a write finished, leaving a writer idle; for SQLite (one writer)
319      * converting and writing would not overlap at all. The extra slot holds one ready batch that waits for a
320      * permit, so the writers stay busy. With one writer this also keeps the SQLite writes in sequence order.
321      *
322      * <p>Memory: at most {@code converters + writers + 3} batches (being filled, converting, writing, one ready).
323      * A failed batch throws out of this method; the other in-flight batches are cancelled.
324      *
325      * @param writers Maximum number of batches written at the same time
326      * @throws IOException Failed to read the index
327      */
328     private void stepScan(int writers) throws IOException {
329         Path tempDir = Files.createTempDirectory("mvn-index");
330 
331         try (
332             ResourceHandler remote = new UriResourceHandler(this.indexFolder);
333             WritableResourceHandler local = new PathWritableResourceHandler(tempDir);
334             IndexReader indexReader = new IndexReader(local, remote)
335         ) {
336             this.indexId = indexReader.getIndexId();
337             if (this.dbType == DatabaseType.OPENSEARCH) {
338                 // The index is named after the Maven index id, known only now
339                 this.databaseRepository.createIndexOpenSearch(this.indexId);
340             }
341             this.summary.putIndex("indexId", indexReader.getIndexId());
342             this.summary.setIndexTimestamp(toInstant(indexReader.getPublishedTimestamp()));
343             this.summary.putIndex("incremental", indexReader.isIncremental());
344             this.summary.putIndex("chunks", indexReader.getChunkNames().size());
345             LOG.log(Level.INFO,"indexRepoId=" + indexReader.getIndexId());
346             LOG.log(Level.INFO,"indexLastPublished=" + indexReader.getPublishedTimestamp());
347             LOG.log(Level.INFO,"isIncremental=" + indexReader.isIncremental());
348             LOG.log(Level.INFO,"indexRequiredChunkNames=" + indexReader.getChunkNames());
349 
350             int batchSize = switch (this.dbType) {
351                 case MYSQL -> this.configMgr.getMysqlBatchSize();
352                 case MARIADB -> this.configMgr.getMariadbBatchSize();
353                 case PSQL -> this.configMgr.getPsqlBatchSize();
354                 case SQLITE -> this.configMgr.getSqliteBatchSize();
355                 case DUCKDB -> this.configMgr.getDuckdbBatchSize();
356                 case MONGODB -> this.configMgr.getMongodbBatchSize();
357                 case OPENSEARCH -> this.configMgr.getOpensearchBatchSize();
358             };
359 
360             // Conversion is CPU bound; 2-8 threads keep up with the single decoding thread
361             int converters = Math.clamp(Runtime.getRuntime().availableProcessors() / 2, 2, 8);
362             LOG.log(Level.INFO, "Concurrent batch converters: {0}", converters);
363             this.summary.putSetting("batchSize", batchSize);
364             this.summary.putSetting("converters", converters);
365 
366             this.recordSeq = 0;
367             long stored;
368             try (Stream<ChunkReader> chunks = StreamSupport.stream(indexReader.spliterator(), false)) {
369                 stored = chunks
370                     .flatMap(this::chunkRecords)
371                     .gather(Gatherers.windowFixed(batchSize))
372                     .gather(Gatherers.mapConcurrent(converters, this::convertBatch))
373                     .filter(batch -> batch.size() > 0)
374                     .gather(Gatherers.mapConcurrent(writers + 1, this::storeBatch))
375                     .mapToLong(Integer::longValue)
376                     .sum();
377             }
378             LOG.log(Level.INFO, "Scanned {0} records, stored {1} records", new Object[]{this.recordSeq, stored});
379             this.summary.setRecords(this.recordSeq, stored);
380         }
381     }

393     /**
394      * Stream the raw records of one chunk, numbered in index order; the chunk is closed when its stream is consumed.
395      */
396     private Stream<RawRecord> chunkRecords(ChunkReader chunkReader) {
397         LOG.log(Level.INFO,"  chunkName=" + chunkReader.getName());
398         LOG.log(Level.INFO,"  chunkVersion=" + chunkReader.getVersion());
399         LOG.log(Level.INFO,"  chunkPublished=" + chunkReader.getTimestamp());
400         this.summary.addIndexChunk(chunkReader.getName(), chunkReader.getVersion(), toInstant(chunkReader.getTimestamp()));
401 
402         return StreamSupport.stream(chunkReader.spliterator(), false)
403             .map(data -> new RawRecord(++this.recordSeq, data))
404             .onClose(() -> {
405                 try {
406                     chunkReader.close();
407                 } catch (IOException e) {
408                     throw new UncheckedIOException(e);
409                 }
410             });
411     }

413     /**
414      * Convert one batch of raw records. Runs on a virtual thread started by {@code mapConcurrent}.
415      *
416      * @param raws Raw records, in sequence order
417      * @return Batch ready to write; empty when every record was skipped
418      */
419     private Batch convertBatch(List<RawRecord> raws) {
420         // RecordExpander has no state, one per batch keeps it obviously thread confined
421         RecordExpander recordExpander = new RecordExpander();
422         Map<String, Integer> maxLengths = new HashMap<>();
423 
424         List<Document> documents = new ArrayList<>(raws.size());
425         for (RawRecord raw : raws) {
426             Document document = toDocument(raw.seq(), recordExpander.apply(raw.data()), maxLengths);
427             if (document != null) {
428                 documents.add(document);
429             }
430         }
431         maxLengths.forEach((key, length) -> this.maxLengthTracker.merge(key, length, Integer::max));
432 
433         if (documents.isEmpty()) {
434             return new Batch(0, 0, null, null, null);
435         }
436         // Read before preparing SQL rows: that removes fields from the documents
437         long counter = documents.getLast().getLong(DatabaseRepository.JSON_FIELD_ID);
438         if (this.dbType == DatabaseType.MONGODB) {
439             return new Batch(counter, documents.size(), documents, null, null);
440         }
441         if (this.dbType == DatabaseType.OPENSEARCH) {
442             // Encode here, on the conversion thread, so the writer only sends bytes
443             return new Batch(counter, documents.size(), null, null, OpenSearchClient.toBulkData(documents));
444         }
445 
446         List<DatabaseRepository.SqlRow> rows = new ArrayList<>(documents.size());
447         try {
448             for (Document document : documents) {
449                 rows.add(this.databaseRepository.toSQLRow(document));
450             }
451         } catch (SQLException e) {
452             throw new IllegalStateException("SQL row preparation failed for records counter=" + counter, e);
453         }
454         if (this.dbType == DatabaseType.PSQL || this.dbType == DatabaseType.MYSQL || this.dbType == DatabaseType.MARIADB) {
455             // Encode here, on the conversion thread, so the writer only streams bytes
456             return new Batch(counter, rows.size(), null, null, DatabaseRepository.toCopyData(rows, this.dbType));
457         }
458         return new Batch(counter, rows.size(), null, rows, null);
459     }

461     /**
462      * Convert one index record to a JSON document.
463      *
464      * @param seq Record sequence number, the primary key
465      * @param record Expanded index record
466      * @param maxLengths Max string length per key, for this batch
467      * @return The document, or <code>null</code> when the record has no version and is skipped
468      */
469     private Document toDocument(long seq, Record record, Map<String, Integer> maxLengths) {
470         Document jsonDoc = new Document(DatabaseRepository.JSON_FIELD_ID, seq);
471         record.getExpanded().forEach((k, v) -> {
472             if (k.getProto().equals(String.class)) {
473                 String value = strip(record.getString(k));
474                 jsonDoc.append(k.getName(), value);
475 
476                 // Track the maximum length of string values for each key for logging and potential future optimizations
477                 if (value != null) {
478                     maxLengths.merge(k.getName(), value.length(), Integer::max);
479                 }
480             } else if (k.getProto().equals(String[].class)) {
481                 List<String> stringList = new ArrayList<>();
482                 for (String s : record.getStringArray(k)) {
483                     stringList.add(strip(s));
484                 }
485                 jsonDoc.append(k.getName(), stringList);
486             } else if (k.getProto().equals(Long.class)) {
487                 jsonDoc.append(k.getName(), record.getLong(k));
488             } else if (k.getProto().equals(Boolean.class)) {
489                 jsonDoc.append(k.getName(), record.getBoolean(k));
490             } else {
491                 LOG.log(Level.WARNING,"Unrecognized key type: " + k + "=" + v + ", name=" + k.getName() + ", type=" + v.getClass().getSimpleName());
492             }
493         });
494 
495         // Parse version
496         String versionString = record.getString(Record.VERSION);
497         if (versionString == null || versionString.isBlank()) {
498             // DESCRIPTOR, ALL_GROUPS and ROOT_GROUPS are index metadata and never have a version
499             Level level = record.getType() == Record.Type.ARTIFACT_ADD ? Level.WARNING : Level.FINE;
500             LOG.log(level, "Record without version found, skipping: {0}", record.getType());
501             return null;
502         }
503         VersionAnalyser analyzedVersion = new VersionAnalyser(versionString);
504         jsonDoc.append(VersionAnalyser.KEY_MAJOR_VERSION, analyzedVersion.majorVersion());
505         jsonDoc.append(VersionAnalyser.KEY_VERSION_SEQ, analyzedVersion.versionSeq());
506 
507         // Calculate file name
508         String classifier = record.getString(Record.CLASSIFIER);
509         String artifactId = record.getString(Record.ARTIFACT_ID);
510         String fileExtension = record.getString(Record.FILE_EXTENSION);
511 
512         String fileName = (classifier == null || classifier.isEmpty())
513                 ? artifactId + "-" + versionString + "." + fileExtension
514                 : artifactId + "-" + versionString + "-" + classifier + "." + fileExtension;
515         jsonDoc.append(DatabaseRepository.JSON_FIELD_FILE_NAME, fileName);
516 
517         return jsonDoc;
518     }

520     /**
521      * Write one batch to the database. Runs on a virtual thread started by {@code mapConcurrent}.
522      *
523      * @param batch Converted batch to write
524      * @return Number of records written
525      */
526     private Integer storeBatch(Batch batch) {
527         try {
528             this.writerPermits.acquire();
529         } catch (InterruptedException e) {
530             // Cancelled because another batch failed
531             Thread.currentThread().interrupt();
532             throw new IllegalStateException("Interrupted while waiting to store records counter=" + batch.counter(), e);
533         }
534         try {
535             switch (this.dbType) {
536                 case SQLITE -> this.databaseRepository.storeSQL(batch.rows(), batch.counter());
537                 case DUCKDB -> this.databaseRepository.storeDuckDB(batch.rows(), batch.counter());
538                 case MYSQL, MARIADB, PSQL -> this.databaseRepository.storeCopyData(batch.payload(), batch.size(), batch.counter());
539                 case MONGODB -> this.databaseRepository.storeMongoDB(batch.documents(), batch.counter(), this.indexId);
540                 case OPENSEARCH -> this.databaseRepository.storeOpenSearch(batch.payload(), batch.size(), batch.counter(), this.indexId);
541             }
542         } finally {
543             this.writerPermits.release();
544         }
545         return batch.size();
546     }
