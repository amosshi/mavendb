48      /**
49       * Integer settings: property key, default value, and valid range [min, max].
50       * A missing, blank, non-numeric or out-of-range value falls back to the default.
51       */
52      enum IntSetting {
53          MYSQL_BATCH_SIZE("org.mavendb.mysql.batch.size", 50_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
54          MARIADB_BATCH_SIZE("org.mavendb.mariadb.batch.size", 50_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
55          PSQL_BATCH_SIZE("org.mavendb.psql.batch.size", 50_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
56          /** For SQLite, 100k and 50k has similar performance. */
57          SQLITE_BATCH_SIZE("org.mavendb.sqlite.batch.size", 100_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
58          DUCKDB_BATCH_SIZE("org.mavendb.duckdb.batch.size", 100_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
59          MONGODB_BATCH_SIZE("org.mavendb.mongodb.batch.size", 20_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
60          /** 10k documents is a bulk request of about 13 MB, within the 5-15 MB OpenSearch recommends. */
61          OPENSEARCH_BATCH_SIZE("org.mavendb.opensearch.batch.size", 10_000, MIN_BATCH_SIZE, MAX_BATCH_SIZE),
62          /** Maximum concurrent writer threads; 4 gives good throughput without overwhelming the database. */
63          THREAD_POOL_SIZE("org.mavendb.thread.pool.size", 4, 2, 64);
64  
65          final String key;
66          final int defaultValue;
67          final int min;
68          final int max;
69  
70          IntSetting(String key, int defaultValue, int min, int max) {
71              this.key = key;
72              this.defaultValue = defaultValue;
73              this.min = min;
74              this.max = max;
75          }
76      }

90      /**
91       * Get database user from configuration with logging for blank values.
92       *
93       * @param configKey Configuration property key
94       * @return Database user name or empty string if not configured
95       */
96      private String getDatabaseUser(String configKey) {
97          String user = config.getProperty(configKey);
98          if (user == null || user.isBlank()) {
99              LOG.log(Level.WARNING, "{0} user not configured, using empty string", configKey);
100             return "";
101         }
102         return user;
103     }

120     /**
121      * Parse and validate an integer setting from configuration, falling back to its default.
122      *
123      * @param setting Setting to parse
124      * @return Validated value
125      */
126     private int parseIntSetting(IntSetting setting) {
127         String valueStr = config.getProperty(setting.key);
128         if (valueStr == null || valueStr.isBlank()) {
129             LOG.log(Level.INFO, "{0} not configured, using default: {1}",
130                     new Object[]{setting.key, setting.defaultValue});
131             return setting.defaultValue;
132         }
133 
134         try {
135             int value = Integer.parseInt(valueStr.trim());
136             if (value < setting.min || value > setting.max) {
137                 LOG.log(Level.WARNING,
138                         "{0} value {1} is out of valid range [{2}, {3}], using default: {4}",
139                         new Object[]{setting.key, value, setting.min, setting.max, setting.defaultValue});
140                 return setting.defaultValue;
141             }
142             LOG.log(Level.INFO, "{0} configured: {1}", new Object[]{setting.key, value});
143             return value;
144         } catch (NumberFormatException _) {
145             LOG.log(Level.WARNING, "{0} configuration invalid: {1}, using default: {2}",
146                     new Object[]{setting.key, valueStr, setting.defaultValue});
147             return setting.defaultValue;
148         }
149     }

151     /**
152      * Constructor.
153      *
154      * @param config Properties object with configuration values (will be defensively copied)
155      */
156     public ConfigurationManager(Properties config) {
157         // Create a defensive copy to prevent external mutation of configuration
158         this.config = new Properties();
159         if (config != null) {
160             this.config.putAll(config);
161         }
162 
163         // Initialize MySQL connection properties with defaults
164         mysqlConnectionProps.setProperty("allowPublicKeyRetrieval", "true");
165         mysqlConnectionProps.setProperty("characterEncoding", "UTF-8");
166         mysqlConnectionProps.setProperty("connectionCollation", "utf8mb4_bin");
167         // Batches are bulk loaded with LOAD DATA LOCAL INFILE, streamed from memory (see DatabaseRepository.storeCopyData).
168         // The driver only honours the in-memory stream with allowLoadLocalInfile=true; allowLoadLocalInfileInPath
169         // ignores it and opens a real file. It lets the server ask for client files: connect only to your own server.
170         // The server side needs --local_infile=1, see docker/compose-mysql.yml.
171         mysqlConnectionProps.setProperty("allowLoadLocalInfile", "true");
172         mysqlConnectionProps.setProperty("useLocalSessionState", "true");
173         mysqlConnectionProps.setProperty("sslMode", "DISABLED");
174 
175         mysqlConnectionProps.setProperty("user", this.getDatabaseUser(CONFIG_MYSQL_USER));
176         mysqlConnectionProps.setProperty("password", this.getDatabasePassword(CONFIG_MYSQL_PASSWORD));
177 
178         // Initialize MariaDB connection properties (MariaDB Connector/J; utf8mb4 and no TLS are its defaults)
179         // Batches are bulk loaded with LOAD DATA LOCAL INFILE, streamed from memory, as for MySQL.
180         // It lets the server ask for client files: connect only to your own server.
181         // The server side needs --local_infile=1, see docker/compose-mariadb.yml.
182         mariadbConnectionProps.setProperty("allowLocalInfile", "true");
183 
184         mariadbConnectionProps.setProperty("user", this.getDatabaseUser(CONFIG_MARIADB_USER));
185         mariadbConnectionProps.setProperty("password", this.getDatabasePassword(CONFIG_MARIADB_PASSWORD));
186 
187         // Initialize PSQL connection properties with defaults
188         psqlConnectionProps.setProperty("ssl", "false");
189         psqlConnectionProps.setProperty("synchronous_commit", "off");  // Improve performance by disabling synchronous commit
190 
191         psqlConnectionProps.setProperty("user", this.getDatabaseUser(CONFIG_PSQL_USER));
192         psqlConnectionProps.setProperty("password", this.getDatabasePassword(CONFIG_PSQL_PASSWORD));
193 
194         // Initialize SQLite connection properties with performance optimizations
195         //
196         // 8GB (8589934592) slowdown the write performance significantly, so we use 
197         // 64MB (64000) and 640MB (64000) has the same performance
198         // So we use 64MB to save memory
199         sqliteConnectionProps.setProperty("cache_size", "-64000");
200         // OFF mode is faster than WAL for single-writer scenarios, but can lead to database corruption on crashes, use with caution
201         sqliteConnectionProps.setProperty("journal_mode", "OFF");
202         // Enable memory-mapped I/O for faster reads - 1GB (1073741824) and 48GB (51539607552) has the same write performance, while create index is slower for 1GB 
203         sqliteConnectionProps.setProperty("mmap_size", "1073741824");
204         //
205         // This is the most important setting for write performance: it controls the size of the database pages. 
206         // Larger page size can significantly improve write performance for large batch inserts, but it also increases memory usage and can lead to fragmentation. 
207         // 64KB is a good balance for our use case, as it provides much better performance than the default 4KB without excessive memory usage. 
208         // 
209         // Set page size to 64KB for optimal performance: it is faster than the default 4KB for data-refersh.sql: 479 vs 1,311 seconds as of Feb 2026
210         // Note: page size must be set before database creation, so it is not configurable at runtime.
211         sqliteConnectionProps.setProperty("page_size", "65536");
212         // Disable synchronous mode for faster writes (use with caution as it can lead to data loss on crashes)
213         sqliteConnectionProps.setProperty("synchronous", "OFF");
214         // Increase temp store to memory for faster operations
215         sqliteConnectionProps.setProperty("temp_store", "MEMORY");
216 
217         // Initialize DuckDB connection properties: DuckDB settings, applied when the database file is opened
218         //
219         // DuckDB runs in this JVM process, next to the -Xmx heap: cap its memory (default: 80% of the RAM).
220         // Aggregates larger than this spill to a temp directory next to the database file.
221         duckdbConnectionProps.setProperty("memory_limit", "32GB");
222 
223         // Parse and validate all integer settings once
224         for (IntSetting setting : IntSetting.values()) {
225             this.intSettings.put(setting, parseIntSetting(setting));
226         }
227    }
