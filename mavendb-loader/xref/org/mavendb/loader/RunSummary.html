26  /**
27   * Machine readable summary of one execution, written as <code>log/&lt;timestamp&gt;-&lt;dbtype&gt;.json</code>
28   * next to the text log of the same run.
29   *
30   * <p>Written for failed runs too, with {@code status} {@code FAILED} and the error, so every run leaves a record.
31   * All durations are in milliseconds. Filled in on the main thread only.
32   */
33  final class RunSummary {
34  
35      private static final Logger LOG = Logger.getLogger(RunSummary.class.getName());
36  
37      private static final JsonWriterSettings JSON_SETTINGS = JsonWriterSettings.builder()
38          .outputMode(JsonMode.RELAXED)
39          .indent(true)
40          .build();

42      private static final double BYTES_PER_GIB = 1024d * 1024 * 1024;

52      private Long databaseSizeBytes;

57      private final Document settings = new Document();

60      private Map<String, Integer> maxFieldLengths = Map.of();

66       * Start the summary of a run.

78      /**
79       * Mask credentials in a connection URL: the user info of {@code scheme://user:password@host}
80       * and {@code password=} parameters.
81       */
82      static String maskUrl(String url) {
83          if (url == null) {
84              return null;
85          }
86          return url.replaceAll("(//)[^/@]*@", "$1***@")
87                    .replaceAll("(?i)(password=)[^&;]*", "$1***");
88      }
89  
90      void setDatabaseUrl(String url) {
91          this.databaseUrl = maskUrl(url);
92      }

131     private Document database() {
132         Document db = new Document("url", this.databaseUrl);
133         if (this.databaseSizeBytes != null) {
134             db.append("sizeBytes", this.databaseSizeBytes)
135               .append("sizeGiB", Math.round(this.databaseSizeBytes * 10.0 / BYTES_PER_GIB) / 10.0);
136         }
137         return db;
138     }

168     /**
169      * Build the JSON document. Field order is the order a reader looks for things: outcome, timing, volume, details.
170      */
171     Document toDocument() {
172         OffsetDateTime endTime = OffsetDateTime.now();
173         long totalMillis = Duration.between(this.startTime, endTime).toMillis();
174 
175         Document doc = new Document()
176             .append("startTime", this.startTime.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME))
177             .append("endTime", endTime.format(DateTimeFormatter.ISO_OFFSET_DATE_TIME))
178             .append("status", this.status);
179         if (this.error != null) {
180             doc.append("error", this.error);
181         }
182         doc.append("dbType", this.dbType.name().toLowerCase());
183 
184         // When the Maven Central index data was published, and how old it was when this run started
185         doc.append("indexTimestamp", this.indexTimestamp == null ? null : this.indexTimestamp.toString());
186         if (this.indexTimestamp != null) {
187             double ageDays = Duration.between(this.indexTimestamp, this.startTime.toInstant()).toMinutes() / 1_440.0;
188             doc.append("indexAgeDays", Math.round(ageDays * 10) / 10.0);
189         }
190 
191         Document durationsMs = new Document("total", totalMillis);
192         durationsMs.putAll(this.durations);
193         doc.append("durationsMs", durationsMs)
194            .append("totalMinutes", Math.round(totalMillis / 6_000.0) / 10.0);
195 
196         Document records = new Document()
197             .append("scanned", this.recordsScanned)
198             .append("stored", this.recordsStored)
199             .append("skipped", this.recordsScanned - this.recordsStored);
200         Long importMillis = this.durations.getLong("import");
201         if (importMillis != null && importMillis > 0) {
202             records.append("importPerSecond", Math.round(this.recordsStored * 1000.0 / importMillis));
203         }
204         doc.append("records", records);
205         if (!this.tableRows.isEmpty()) {
206             doc.append("tableRows", this.tableRows);
207         }
208 
209         doc.append("index", new Document("source", this.reposFolder).append("details", this.index).append("chunks", this.indexChunks))
210            .append("database", database())
211            .append("settings", this.settings)
212            .append("maxFieldLengths", new Document(new LinkedHashMap<>(this.maxFieldLengths)))
213            .append("build", buildInfo())
214            .append("environment", environment())
215            .append("logFile", this.logFile);
216         return doc;
217     }
218 
219     String toJson() {
220         return toDocument().toJson(JSON_SETTINGS);
221     }
222 
223     /**
224      * Write the summary; a failure is logged, never thrown, so it cannot hide the outcome of the run.
225      *
226      * @param file Target JSON file
227      */
228     void write(Path file) {
229         try {
230             Files.writeString(file, toJson() + System.lineSeparator(), StandardCharsets.UTF_8);
231             LOG.log(Level.INFO, "Run summary: {0}", file);
232         } catch (IOException | RuntimeException e) {
233             LOG.log(Level.WARNING, "Failed to write the run summary " + file, e);
234         }
235     }

237     /** Version, git revision and build time from the manifest of mavendb-loader.jar; empty when not run from the jar. */
238     private static Document buildInfo() {
239         Document build = new Document();
240         String codeSource = RunSummary.class.getProtectionDomain().getCodeSource().getLocation().getPath();
241         if (!codeSource.endsWith(".jar")) {
242             return build;
243         }
244         try (JarFile jar = new JarFile(codeSource)) {
245             Manifest manifest = jar.getManifest();
246             if (manifest == null) {
247                 return build;
248             }
249             Attributes versions = manifest.getAttributes("Versions");
250             if (versions != null) {
251                 build.append("version", versions.getValue("Implementation-Version"))
252                      .append("gitRevision", versions.getValue("Implementation-SCM-Revision"))
253                      .append("gitBranch", versions.getValue("Implementation-SCM-Branch"));
254             }
255             build.append("buildTime", manifest.getMainAttributes().getValue("Build-Time"));
256         } catch (IOException e) {
257             LOG.log(Level.FINE, "Cannot read the jar manifest", e);
258         }
259         return build;
260     }
261 
262     private static Document environment() {
263         Runtime runtime = Runtime.getRuntime();
264         return new Document()
265             .append("javaVersion", Runtime.version().toString())
266             .append("javaVm", System.getProperty("java.vm.name"))
267             .append("os", System.getProperty("os.name") + " " + System.getProperty("os.version") + " " + System.getProperty("os.arch"))
268             .append("availableProcessors", runtime.availableProcessors())
269             .append("maxHeapBytes", runtime.maxMemory())
270             .append("jvmUptimeMs", ManagementFactory.getRuntimeMXBean().getUptime());
271     }
