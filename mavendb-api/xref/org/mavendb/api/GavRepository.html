1   package org.mavendb.api;

3   import java.sql.ResultSet;
4   import java.sql.SQLException;
5   import java.util.List;
6   import java.util.Optional;
7   import org.springframework.jdbc.core.simple.JdbcClient;
8   import org.springframework.stereotype.Repository;
9   
10  /**
11   * All SQL of the API. Every query is an indexed lookup on the loader's SQLite schema
12   * (<code>db/sqlite/create.sql</code> and <code>data-refresh.sql</code> in mavendb-loader).
13   */
14  @Repository
15  public class GavRepository {
16  
17      /** Rows returned at most by a lookup by file name or sha1. */
18      static final int MAX_FILE_MATCHES = 100;
19  
20      private static final String GAV_COLUMNS = """
21          group_id, artifact_id, artifact_version, classifier, packaging, file_extension,
22          file_name, sha1, file_size, file_modified
23          """;
24  
25      private final JdbcClient jdbc;
26  
27      GavRepository(JdbcClient jdbc) {
28          this.jdbc = jdbc;
29      }
30  
31      /**
32       * Files with this name; uses index <code>idx_gav_file_name</code>.
33       * The same name can exist in several groups, like <code>core-1.0.jar</code>.
34       */
35      List<Gav> findByFileName(String fileName) {
36          return this.jdbc.sql("SELECT " + GAV_COLUMNS + """
37                  FROM gav WHERE file_name = ?
38                  ORDER BY group_id, artifact_id, version_seq LIMIT ?
39                  """)
40              .params(fileName, MAX_FILE_MATCHES)
41              .query(GavRepository::gav)
42              .list();
43      }
44  
45      /** Files with this sha1; uses index <code>idx_gav_sha1</code>. */
46      List<Gav> findBySha1(String sha1) {
47          return this.jdbc.sql("SELECT " + GAV_COLUMNS + """
48                  FROM gav WHERE sha1 = ?
49                  ORDER BY group_id, artifact_id, version_seq LIMIT ?
50                  """)
51              .params(sha1, MAX_FILE_MATCHES)
52              .query(GavRepository::gav)
53              .list();
54      }
55  
56      /** Artifacts of a group, from the summary table <code>ga</code> (index <code>idx_ga</code>). */
57      List<Artifact> findArtifacts(String groupId, int limit, int offset) {
58          return this.jdbc.sql("""
59                  SELECT artifact_id, artifact_version_counter, file_modified_max
60                  FROM ga WHERE group_id = ?
61                  ORDER BY artifact_id LIMIT ? OFFSET ?
62                  """)
63              .params(groupId, limit, offset)
64              .query((rs, i) -> new Artifact(
65                  rs.getString("artifact_id"),
66                  rs.getLong("artifact_version_counter"),
67                  nullableLong(rs, "file_modified_max")))
68              .list();
69      }
70  
71      /**
72       * Highest version of an artifact by <code>version_seq</code> (the loader's sortable version number,
73       * so 3.10 sorts after 3.9); ties, like 2.0-RC1 and 2.0, go to the newer file. Uses index <code>idx_gav</code>.
74       */
75      Optional<LatestVersion> findLatest(String groupId, String artifactId) {
76          return this.jdbc.sql("""
77                  SELECT group_id, artifact_id, artifact_version, file_modified
78                  FROM gav WHERE group_id = ? AND artifact_id = ?
79                  ORDER BY version_seq DESC, file_modified DESC LIMIT 1
80                  """)
81              .params(groupId, artifactId)
82              .query((rs, i) -> new LatestVersion(
83                  rs.getString("group_id"),
84                  rs.getString("artifact_id"),
85                  rs.getString("artifact_version"),
86                  nullableLong(rs, "file_modified")))
87              .optional();
88      }
89  
90      private static Gav gav(ResultSet rs, int rowNum) throws SQLException {
91          return new Gav(
92              rs.getString("group_id"),
93              rs.getString("artifact_id"),
94              rs.getString("artifact_version"),
95              rs.getString("classifier"),
96              rs.getString("packaging"),
97              rs.getString("file_extension"),
98              rs.getString("file_name"),
99              rs.getString("sha1"),
100             nullableLong(rs, "file_size"),
101             nullableLong(rs, "file_modified"));
102     }
103 
104     private static Long nullableLong(ResultSet rs, String column) throws SQLException {
105         long value = rs.getLong(column);
106         return rs.wasNull() ? null : value;
107     }
108 }
