16  /** Tomcat's rate limit filter as wired by {@link RateLimitConfig}: applies to /v1, not to the health check. */
17  @SpringBootTest(properties = "mavendb.api.rate-limit.per-client-per-minute=2")
18  @AutoConfigureMockMvc
19  class RateLimitIntegrationTest {
20  
21      @Autowired
22      private MockMvc mvc;
23  
24      @DynamicPropertySource
25      static void sqlite(DynamicPropertyRegistry registry) {
26          registry.add("mavendb.api.sqlite.path", () -> TestDatabase.file().toString());
27      }
28  
29      @Test
30      void apiIsLimitedHealthIsNot() throws Exception {
31          String url = "/v1/groups/org.apache.commons/artifacts/commons-lang3/latest";
32          this.mvc.perform(get(url))
33              .andExpect(status().isOk())
34              .andExpect(header().string("RateLimit-Policy", containsString("\"per-ip\";q=2;w=60")))
35              .andExpect(header().string("RateLimit", containsString("\"per-ip\";r=1")));
36          this.mvc.perform(get(url)).andExpect(status().isOk());
37          this.mvc.perform(get(url)).andExpect(status().isTooManyRequests());
38  
39          for (int i = 0; i < 5; i++) {
40              this.mvc.perform(get("/actuator/health")).andExpect(status().isOk());
41          }
42      }
