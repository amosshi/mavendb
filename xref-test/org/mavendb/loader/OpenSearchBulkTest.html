20      @Test
21      @DisplayName("Each document is an index action with its _id, then the source without _id")
22      void testBulkData() {
23          Document first = new Document("_id", 42L).append("groupId", "org.example").append("fileSize", 3_000_000_000L)
24              .append("hasSources", true).append("name", "tab\there \"quoted\" 例子 😀");
25          Document second = new Document("_id", 43L).append("groupId", "g").append("classNames", List.of("a.B", "c.D"));
26  
27          String body = new String(OpenSearchClient.toBulkData(List.of(first, second)), StandardCharsets.UTF_8);
28          String[] lines = body.split("\n", -1);
29          assertEquals(5, lines.length, "two documents, two lines each, newline terminated");
30          assertEquals("", lines[4]);
31  
32          assertEquals("{\"index\":{\"_id\":\"42\"}}", lines[0]);
33          Document source = Document.parse(lines[1]);
34          assertFalse(source.containsKey("_id"), "_id is metadata, not allowed in the source");
35          assertEquals("org.example", source.getString("groupId"));
36          assertEquals(3_000_000_000L, source.get("fileSize", Number.class).longValue());
37          assertEquals("tab\there \"quoted\" 例子 😀", source.getString("name"));
38          assertFalse(lines[1].contains("$numberLong"), "relaxed JSON: plain numbers");
39  
40          assertEquals("{\"index\":{\"_id\":\"43\"}}", lines[2]);
41          assertEquals(List.of("a.B", "c.D"), Document.parse(lines[3]).getList("classNames", String.class));
42      }

44      @Test
45      @DisplayName("Bulk bodies split at document boundaries, under the size limit; an oversized document goes alone")
46      void testSplitBulk() {
47          List<Document> documents = new java.util.ArrayList<>();
48          for (int i = 0; i < 20; i++) {
49              // Document 7 is larger than the limit; the others vary, with escaped newlines in the text
50              int length = i == 7 ? 5_000 : 100 + i * 37;
51              documents.add(new Document("_id", (long) i).append("text", "x\n".repeat(length / 2)));
52          }
53          byte[] data = OpenSearchClient.toBulkData(documents);
54          int maxBytes = 2_000;
55  
56          List<int[]> ranges = OpenSearchClient.splitBulk(data, maxBytes);
57          int expectedStart = 0;
58          int docs = 0;
59          for (int[] range : ranges) {
60              assertEquals(expectedStart, range[0], "ranges are contiguous");
61              String part = new String(data, range[0], range[1] - range[0], StandardCharsets.UTF_8);
62              String[] lines = part.split("\n");
63              assertEquals(0, lines.length % 2, "whole documents only");
64              assertTrue(lines[0].startsWith("{\"index\""));
65              assertTrue(part.endsWith("\n"));
66              assertTrue(range[1] - range[0] <= maxBytes || lines.length == 2, "over the limit only for a single document");
67              docs += lines.length / 2;
68              expectedStart = range[1];
69          }
70          assertEquals(data.length, expectedStart, "ranges cover the whole body");
71          assertEquals(20, docs);
72          assertTrue(ranges.size() > 3);
73  
74          assertEquals(1, OpenSearchClient.splitBulk(data, Integer.MAX_VALUE).size(), "a small body is one request");
75          assertTrue(OpenSearchClient.splitBulk(new byte[0], maxBytes).isEmpty());
76      }
